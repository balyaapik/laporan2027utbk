      {\color{StudentGray}\fontsize{12}{15}\selectfont Karena itu, tindak lanjut tidak boleh hanya berdasarkan skor akhir.}

      {\color{StudentBlue}\bfseries\fontsize{11}{13.5}\selectfont Yang dianalisis:}\par

  \end{columns}

\begin{frame}{Apa yang ingin dicapai melalui pembimbingan?}
  \kicker{Tujuan Program}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.31\textwidth}
      \begin{block}{Terarah}

      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{Terukur}
        Perkembangan terlihat dari diagnostik, latihan, tryout, dan tindak lanjut.
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{Berkelanjutan}
        Sekolah, tentor, siswa, dan orang tua menjaga proses sampai masa seleksi.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.20cm}
  {\fontsize{11.8}{14.8}\selectfont
  \begin{itemize}

    \item Pilihan program studi dibahas dengan mempertimbangkan kemampuan, minat, dan peluang.
  \end{itemize}}
\end{frame}
